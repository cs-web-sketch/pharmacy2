// Initialize Grid function initGrid() { grid.innerHTML = ''; for (let i = startNum; i <= endNum; i++) { const card = document.createElement('div'); card.className = 'num-card'; card.innerText = i; // Pre-set styles based on current number if (i < currentNum) { card.classList.add('status-done'); } else if (i === currentNum) { card.classList.add('status-calling'); mainNumber.innerText = i; } grid.appendChild(card); cardElements[i] = card; } } // Call the next number function callNextNumber() { // Mark previous as done if (cardElements[currentNum]) { cardElements[currentNum].classList.remove('status-calling'); cardElements[currentNum].classList.add('status-done'); } // Increment or reset currentNum++; if (currentNum > endNum) { // Loop back for demo purposes currentNum = startNum; // Reset all cards to default visually Object.values(cardElements).forEach(card => { card.className = 'num-card'; }); } // Update Main Display with Animation mainNumber.innerText = currentNum; // Re-trigger CSS animation mainNumber.classList.remove('pop-anim'); callingBox.classList.remove('active-border'); // Trigger reflow to restart animation void mainNumber.offsetWidth; mainNumber.classList.add('pop-anim'); callingBox.classList.add('active-border'); // Highlight new card in grid if (cardElements[currentNum]) { cardElements[currentNum].classList.add('status-calling'); // Scroll grid to the new number smoothly cardElements[currentNum].scrollIntoView({ behavior: 'smooth', block: 'center' }); } } // Setup initGrid(); // Simulate a new call every 6 seconds to show off the design setInterval(callNextNumber, 6000);
